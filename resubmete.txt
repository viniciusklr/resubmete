function getCampanhasAtivas() {
  const tabela = document.querySelectorAll("table")[4];

  return [...tabela.querySelectorAll("tr")]
    .map(tr =>
      [...tr.querySelectorAll("td")].map(td =>
        td.innerText.replace(/\s+/g, " ").trim()
      )
    )
    .filter(l => l.length);
}

function campanhasParaResubmeter() {
  const rows = getCampanhasAtivas();
  let lastCampaignId = null;
  let lastCampaignName = null;

  return rows
    .map(row => {
      const isFull = row.length >= 18;

      if (isFull) {
        lastCampaignId = row[0]?.replace(/\D/g, "");
        lastCampaignName = row[1];
      }

      const campaignId = lastCampaignId;
      const campaignName = lastCampaignName;
      const mailing = isFull ? row[8] : row[0];
      const valor = parseInt(isFull ? row[10] : row[2], 10);

      if (!campaignId || !mailing || isNaN(valor)) return null;

      return {
        campaignId,
        campaignName,
        mailing,
        valor
      };
    })
    .filter(item => item && item.valor < 100);
}

async function executarResubmissao(campanha) {
  try {
    console.log(
      `Enviando campanha ${campanha.campaignId} - ${campanha.campaignName}`
    );

    const response = await fetch(
      "http://172.16.10.4/WebMonitor/Default.aspx/ExecuteFastBuild",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json; charset=utf-8",
        },
        body: JSON.stringify({
          data: {
            Mailing: campanha.mailing,
            CampaignId: campanha.campaignId,
            UserId: "15971",
            CampaignName: campanha.campaignName,
            Dispositions: dispositions,
            resetCustomerAttempts: true,
            resetPhoneNumberAttempts: true,
          },
        }),
      }
    );

    const result = await response.json();

    console.log(
      `API enviada para campanha ${campanha.campaignId}:`,
      result
    );

  } catch (err) {
    console.error(
      `Erro ao chamar API da campanha ${campanha.campaignId}:`,
      err
    );
  }
}

async function monitorarCampanhas() {
  console.log("Verificando campanhas...");

  const campanhas = campanhasParaResubmeter();

  console.log("Campanhas abaixo de 100:", campanhas);

  for (const campanha of campanhas) {
    await executarResubmissao(campanha);
  }
}

monitorarCampanhas();
setInterval(monitorarCampanhas, 30000);
